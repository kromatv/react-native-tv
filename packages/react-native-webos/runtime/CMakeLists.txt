cmake_minimum_required(VERSION 3.24)
project(react_native_webos LANGUAGES C CXX ASM)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

find_program(RN_WEBOS_CCACHE ccache)
if(RN_WEBOS_CCACHE)
  set(CMAKE_C_COMPILER_LAUNCHER "${RN_WEBOS_CCACHE}")
  set(CMAKE_CXX_COMPILER_LAUNCHER "${RN_WEBOS_CCACHE}")
endif()

option(RN_WEBOS_SIMULATOR "Build for the Linux simulator in ../simulator instead of a TV" OFF)
option(RN_WEBOS_TESTS "Build the host's tests, rn-webos-host-tests (simulator builds only)" OFF)
set(RN_WEBOS_UNPORTED "" CACHE STRING "Regular expressions of host sources left out while the port is brought up")
set(RN_WEBOS_UNPORTED_TESTS "" CACHE STRING "Regular expressions of host tests left out while the port is brought up")
option(RN_WEBOS_COVERAGE "Instrument the host and its tests for llvm-cov" OFF)
# A development runtime talks to a dev server (host/devsupport) and has Hermes'
# debugger; a release one carries neither. The simulator only runs development builds.
option(RN_WEBOS_DEVELOPMENT "Build the development runtime: dev server tooling and the debugger" ${RN_WEBOS_SIMULATOR})

set(REACT_NATIVE_DIR "" CACHE PATH "The react-native package the app depends on")
if(NOT EXISTS "${REACT_NATIVE_DIR}/ReactCommon")
  message(FATAL_ERROR "REACT_NATIVE_DIR must be a react-native package, got '${REACT_NATIVE_DIR}'")
endif()

find_package(Threads REQUIRED)

add_compile_definitions(RN_BUILDING)
include("${REACT_NATIVE_DIR}/ReactCommon/cmake-utils/react-native-flags.cmake")

if(NOT RN_WEBOS_CACHE_DIR)
  if(DEFINED ENV{RN_WEBOS_CACHE_DIR})
    set(RN_WEBOS_CACHE_DIR "$ENV{RN_WEBOS_CACHE_DIR}")
  elseif(DEFINED ENV{XDG_CACHE_HOME})
    set(RN_WEBOS_CACHE_DIR "$ENV{XDG_CACHE_HOME}/react-native-webos")
  else()
    set(RN_WEBOS_CACHE_DIR "$ENV{HOME}/.cache/react-native-webos")
  endif()
endif()

if(RN_WEBOS_SIMULATOR)
  include(cmake/Simulator.cmake)
else()
  include(cmake/WebOSPlatform.cmake)
endif()
include(cmake/Dependencies.cmake)
include(cmake/ThirdParty.cmake)
include(cmake/Icu.cmake)
include(cmake/Curl.cmake)
include(cmake/Skia.cmake)
include(cmake/Sdl.cmake)
include(cmake/Hermes.cmake)
include(cmake/Codegen.cmake)
include(cmake/ReactCommon.cmake)
include(cmake/Expo.cmake)
include(cmake/ObjectClosure.cmake)
if(RN_WEBOS_TESTS)
  include(cmake/Tests.cmake)
endif()

add_subdirectory(platform)
add_subdirectory(host)
target_link_libraries(rn_webos_host PUBLIC rn_webos_icuuc)
include(cmake/Libraries.cmake)
